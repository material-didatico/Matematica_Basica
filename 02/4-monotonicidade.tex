\section{Monotonicidade}

%------------------------------------------------------------------------------%
\begin{definicao}{Monotonicidade}{def:Monotonicidade}
  Seja $f$ uma função definida em um intervalo $I$,
  dizemos que
  \begin{description}
  \item[$f$ é crescente]\index{Função!crescente} em $I$ se,
    dados quaisquer $x_1$ e $x_2$ em $I$
    \[
      x_1 < x_2
      \quad\Rightarrow\quad
      f(x_1) < f(x_2);
    \]

  \item[$f$ é decrescente]\index{Função!decrescente} em $I$ se,
    dados quaisquer $x_1$ e $x_2$ em $I$
    \[
      x_1 < x_2
      \quad\Rightarrow\quad
      f(x_1) > f(x_2);
    \]

  \item[$f$ é constante]\index{Função!constante} em $I$ se,
    dados quaisquer $x_1$ e $x_2$ em $I$
    \[
      f(x_1)=f(x_2)
    \]
  \end{description}
\end{definicao}
