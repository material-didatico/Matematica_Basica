\section{Funções:}
As funções surgem quando uma quantidade (Grandeza) depende de outra. Por exemplo, podemos dizer que a área de um quadrado está relacionada ao comprimento do lado, isso quer dizer que uma vez conhecido o comprimento do lado, $x$, conseguimos calcular a área usando a função $f$ descrita pela relação $f(x)=x^2$.
\vspace{0.5cm}

É possível representar uma função de quatro maneiras:

\begin{itemize}
 \item verbalmente (descrevendo-a com palavras)
\hspace{2cm}
\item numericamente (por meio de uma tabela de valores)

\item  visualmente (através de um gráfico)

\item algebricamente (utilizando-se uma fórmula explícita)
\end{itemize}
\vspace{0.5cm}

\begin{defin}(\textbf{Função}): Uma \textbf{função} $f:D\to C$ é uma relação que associa a cada elemento $x$ de um conjunto $D$, chamado \textbf{domínio}, um \textit{único} elemento $f(x)$ (ou $y$) de um conjunto $C$, denominado \textbf{contradomínio}.
\end{defin}

\begin{obs} Quando o domínio de uma função $f$ não for específicado, então o \textbf{domínio} (natural) de $f$ é o conjunto de todos os valores de $x$ para os quais $f$ está definida.
\end{obs}
\begin{obs}
Devemos ter cuidado ao analisarmos o domínio de uma função, quando essa é originada de um problema real. Se, por exemplo, a função $f(x)=x+1$, $f(x)$ representa o valor a ser pago em uma corrida de táxi em função do número de quilômetros rodados $x$, então o domínio de $f$ será $D= \{ x \in \mathbb{R} | x \geq 0 \}$, uma vez que o número de quilômetros rodados é sempre não-negativo.
\end{obs}

Note que o contradomínio $C$ de uma função $f$ pode ser qualquer conjunto que contenha, dentre outros elementos, os valores de $f(x)$. Sendo assim, em muitas situações estamos interessados em trabalhar apenas com o conjunto que contém apenas os valores gerados por $f$, ao qual damos o nome de \textbf{conjunto imagem}.

\begin{defin}(\textbf{Imagem}) Dada uma \textbf{função} $f:D\to C$, o valor $f(x)$ para $x\in D$ é chamado \textbf{imagem de $x$}. O conjunto de todas as imagens da função $f$ é denominado \textbf{conjunto imagem}(ou simplesmente imagem) e denotado por \textbf{\textit{Im}(f)}. Ou seja,  \textit{Im}(f)$=\{f(x);x \in D\}$.
\end{defin}

\begin{defin} (\textbf{Gráfico}) o \textbf{gráfico} de $f$ é o conjunto dos pontos $(x,y)$ do plano cartesiano que satisfazem $y=f(x),$ ou seja,  $G(f)=\{(x, f(x)); x\in D\}$.
\end{defin}

\textbf{Quando que uma aplicação não é função?}
\vspace{0.5cm}

\begin{exe} Verifique se a equação $y^2+x^2-16=0$ permite uma definição de $y$ como função de $x$.
\end{exe}
\textcolor{red}{Inicialmente vamos ver se conseguimos explicitar $y$ como função de $x$.
\begin{eqnarray*}
y^2 &=&16-x^2 \\
y= &\pm &\sqrt{16-x^2}
\end{eqnarray*}
Logo, para cada $x \in \mathbb{R}$, temos dois valores para $y$. Portanto, a equação $y^2+x^2-16=0$ não permite uma definição de $y$ como função de $x$.}
\begin{exe}
Faça o estudo do domínio das seguintes funções \\
a)$f(x)=1-x^4$ \hspace{1cm}    b)$g(x)=\frac{1}{x+4}$  \hspace{1cm}      c)$h(x)=\sqrt{9-x}$  \hspace{1cm} d)$f(t)=\frac{\sqrt{2+t}}{\sqrt{2t-6}}$   \hspace{1cm}  e)$f(x)=\sqrt[3]{2x-3}$  \hspace{1cm} f)p(x)=$\frac{\sqrt{x-1}}{x^2-5x+6}$ \hspace{1cm} \\ %g)$f(x)= \frac{1}{x^2-4}+ \frac{1}{\sqrt{x+4}}$ \\
\textcolor{red}{\textbf{a)} Note que $f$ está bem definida para todo $x \in \mathbb{R}$, logo $D(f)= \mathbb{R}$. \\
\textbf{b)} Nesse caso temos que $g$ está definida quando $x+4 \neq 0 \Rightarrow x \neq -4$, logo $D(g)= \mathbb{R}-\{-4\}$ ou $D(g)= \{ x \in \mathbb{R}| x \neq -4 \}$. \\
\textbf{c)} Para que $h$ seja bem definida $9-x \geq 0 \Rightarrow  x \leq 9 $, logo $D(h)= \{ x \in \mathbb{R}| x \leq 9 \}$. \\
\textbf{d)} Para que $f$ esteja bem definida teremos que ter $2+t \geq 0 \Rightarrow t \geq -2$ e $2t-6 >0 \Rightarrow t>3$, fazendo a interseção desses intervalos temos que $D(f)= \{ x \in \mathbb{R}| x \geq 3 \}$. \\
\textbf{e)} Como o índice a raíz é ímpar, então $D(f)= \mathbb{R}$. \\
\textbf{f)} Para que $p$ esteja bem definida teremos que ter $x-1 \geq 0 \Rightarrow x \geq 0$ e $x^2-5x+6 \neq 0 \Rightarrow x \neq 2$ e $x \neq 3$, portanto $D(p)= \{ x \in \mathbb{R}| x \geq 1, x \neq 2 \hspace{0,15cm x \neq 3} \}$.
}
\end{exe}
\subsection{Simetria}
Se uma função $f$ satisfaz $f (–x)$ = $f (x)$ para todo número $x$ em seu domínio, então $f$ é chamada \textbf{função par}. Por outro lado, se $f$ satisfaz $f (–x) = –f (x)$ para cada número $x$ em seu domínio, uma $f$  é \textbf{função ímpar}.
\begin{exe}
Verifique se as funções abaixo são par, ímpar ou nem par, nem ímpar \\
a)$f(x)=1-x^4$  \hspace{1cm}   b)$g(x)=x^5+x$     \hspace{1cm}    c)$h(x)=2x-x^2$ \\
\textcolor{red}{ \textbf{a)} $f(-x)=1-(-x)^4=1+x^4=f(x)$, logo $f$ é par. \\
\textbf{b)} $g(-x)=(-x)^5-(-x)= -x^5+x=-(x^5-x)=-g(x)$, logo $g$ é ímpar. \\
\textbf{c)} Nesse caso temos que $h(1)=2.1-1^2=1$ e $h(-1)=2(-1)-(-1)^2==-3$, logo $h$ não é nem par, nem ímpar.}
\end{exe}

\subsection{Zeros de função}
Os valores de $x$ que satisfazem a equação $f(x)=0$ são chamados \textbf{zeros} de $f$. Esses valores correspondem aos interceptos-x do gráfico da função.

\begin{exe} Encontre os zeros das seguintes funções:
a) $f(x)=x^3+2x^2-5x$   b) $g(x)= \frac{x^2-4}{x+3}$ \\
\textcolor{red}{\textbf{a)} Para encontrarmos os zeros de $f$ precisamos resolver a equação $x^3+2x^2-5x=0 \Rightarrow x(x^2+2x-5)=0$, portanto $x=0$ ou $x^2+2x-5=0$. Para encontrarmos as raízes de $x^2+2x-5=0$ usamos a fórmula de Bhaskara, uma vez que se trata de uma equação do segundo grau. Com isso, $x^2+2x-5=0 \Rightarrow x=-1+\sqrt{6}$ ou $x=-1-\sqrt{6}$. Logo, $f$ possui três zeros, a saber $x=0$, $x=-1+\sqrt{6}$ e $-1-\sqrt{6}$. \\
\textbf{b)} Para encontrarmos os zeros de $g$ precisamos resolver a equação $\frac{x^2-4}{x+3}=0 \Rightarrow x^2-4=0 \Rightarrow x=2 \hspace{0,3cm} ou \hspace{0,3cm} x=-2$. Logo $g$ possui dois zeros, a saber $x=-2$ e $x=2$. }
\end{exe}

\subsection{Monotonicidade}
%\begin{figure}[H]
%\includegraphics[scale=0.5]{crescente}
%\end{figure}
\begin{defin} Seja $f$ uma função definida em um intervalo $I$, dizemos que
\begin{itemize}
\item[1.] f é \textbf{crescente} em $I$ se, dados quaisquer $x_1$ e $x_2$ em $I$, tais que $x_1< x_2$, tivermos $f(x_1)< f(x_2)$;
\item[2.] f é \textbf{decrescente} em $I$ se, dados quaisquer $x_1$ e $x_2$ em $I$, tais que $x_1< x_2$, tivermos $f(x_1)> f(x_2)$;
\item[3.] f é \textbf{constante} em $I$ se, dados quaisquer $x_1$ e $x_2$ em $I$, tivermos $f(x_1)=f(x_2)$
\end{itemize}
\end{defin}

\subsection{Função Injetora}
\begin{defin}(\textbf{Função Injetora}): Uma função $f$, definida em seu domínio $D$ é \textbf{injetora} quando dados quaisquer valores reais $x_1$, $x_2 \in D$, se $x_1 \neq x_2$, então, $f(x_1) \neq f(x_2)$.
\end{defin}

Em outras palavras, se $f$ é uma função injetora então,
$$f(x_1)=f(x_2) \Leftrightarrow x_1=x_2$$.

\textbf{Teste da reta horizontal:} Uma função é injetora se e somente se nenhuma reta horizontal intercepta seu gráfico mais de uma vez.

\begin{figure}[h!]
\centering
\includegraphics[scale=1.4]{injet}
\caption{Gráfico de uma função que não é injetora, uma vez que $f(x_1)=f(x_2)$ para $x_1 \neq x_2$.}
\end{figure}

\begin{exe} Verifique se as funções abaixo são injetoras \\
a) $f(x)=x^2$ \hspace{1cm}  b)$f(x)=x^3$ \hspace{1cm}   c)$y= \frac{3}{5x-2}$ \hspace{1cm} \\  %d)$f(x)=1-x^2$ \\
\textcolor{red}{\textbf{a)} $f$ não é injetora pois $f(-1)=f(1)=1$. \\
\textbf{b)} Suponha que $f(x_1)=f(x_2) \Rightarrow x_1^3=x_2^3 \Rightarrow x_1^3= \sqrt[3]{x_2^3} \Rightarrow x_1=x_2$, logo $f$ é injetora. \\
\textbf{c)} Suponha que $f(x_1)=f(x_2) \Rightarrow \frac{3}{5x_1-2}= \frac{3}{5x_2-2} \Rightarrow 3(5x_1-2)=3(5x_2-2) \Rightarrow 5x_1-2=5x_2-2 \Rightarrow x_1=x_2.$ Logo, $y$ é injetora. }
\end{exe}

\begin{itemize}
\item[•] O domínio de $f$ é $\mathbb{R}$.
\item[•] Pode ter até $n$ zeros.

\textbf{Exemplo 6: (Função recíproca)}  É uma função do tipo $f(x)=\frac{1}{x^n}$, com $x \neq 0$.
\begin{itemize}

\item[•] o conjunto imagem é $(-\infty,0) \cup (0, \infty)$ se o expoente é ímpar e $(0, \infty)$ se o expoente é par.
\item[•] A função não tem zeros.
\item[•] Quando o expoente é ímpar, a função é decrescente em todos os pontos do domínio. Já quando o expoente é par, $f$ é crescente para $x<0$ e decrescente para $x>0$.
\item[•] Quando o expoente é par, $f$ é par. Quando o expoente é ímpar, $f$ é ímpar.
\end{itemize}

\item[•] A função não possui zeros;
\item[•] se $a>1$, a função é crescente em todo o domínio e se $0<a<1$ a função é decrescente em todo o domínio.
\item[•] A função não é par nem ímpar.
\end{itemize}

\subsection{Função composta}

\begin{defin} \label{com}\textbf{Função composta}] Dadas as funções $f $ e $ g$ cujos domínios são $A$ e $B$, respectivamente, definimos a \textbf{função composta} $f \circ g$ por
$$(f \circ g)(x)= f(g(x))$$
O domínio de $f \circ g$ é o conjunto dos valores de $x \in B$ tais que $g(x) \in A$
\end{defin}

\begin{exe} Se $f(x)= x^2$ e $g(x)=x-3$, encontre as compostas $f \circ g$ e $g \circ f$. \\
\textcolor{red}{Por definição $f \circ g= f(g(x))=f(x-3)=(x-3)^2$ e $g \circ f= g(f(x))=g(x^2)= x^2-3$.}
\end{exe}
\begin{exe} Se $f(x)= \frac{x}{x+1}$, $g(x)=x^{10}$ e $h(x)=x+3$, encontre  $f \circ g \circ h$. \\
\textcolor{red}{$f \circ g \circ h= f(g(h(x)))= f(g(x+3))=f((x+3)^{10})=\frac{(x+3)^{10}}{(x+3)^{10}+1}$.}

\begin{obs} \label{obs} Devemos está atentos ao calcularmos o domínio da função composta. Segundo a definição \ref{com} para que sejamos capazes de calcular $f(g(x))$ é preciso que:
\begin{itemize}

\item[2.] $g(x)$(o valor de $g$ em $x$) pertença ao domínio de $f$;
\end{itemize}
Para deixar mais clara a definição do domínio da função composta, veremos como obtê-lo de forma prática no exemplo a seguir.
\end{obs}

\end{exe}
