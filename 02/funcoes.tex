\section{Funções}

As funções surgem quando uma quantidade (Grandeza) depende de outra. Por
exemplo, podemos dizer que a área de um quadrado está relacionada ao
comprimento do lado, isso quer dizer que uma vez conhecido o comprimento do
lado, $x$, conseguimos calcular a área usando a função $f$ descrita pela
relação $f(x)=x^2$.

É possível representar uma função de quatro maneiras:
\begin{itemize}
  \item verbalmente (descrevendo-a com palavras)
  \item numericamente (por meio de uma tabela de valores)
  \item visualmente (através de um gráfico)
  \item algebricamente (utilizando-se uma fórmula explícita)
\end{itemize}

\begin{definicao}{Função}{def:funcao}
  Uma \emph{função} $f:D\to C$ é uma relação que associa a cada elemento $x$
  de um conjunto $D$, chamado \emph{domínio}, um \emph{único} elemento
  $f(x)$ (ou $y$) de um conjunto $C$, denominado \emph{contradomínio}.
\end{definicao}

\begin{observation}
  Quando o domínio de uma função $f$ não for específicado, então o
  \emph{domínio} (natural) de $f$ é o conjunto de todos os valores de $x$
  para os quais $f$ está definida.
\end{observation}

\begin{observation}
  Devemos ter cuidado ao analisarmos o domínio de uma função, quando essa é
  originada de um problema real.
  Se, por exemplo, a função $f(x)=x+1$, $f(x)$
  representa o valor a ser pago em uma corrida de táxi em função do número de
  quilômetros rodados $x$, então o domínio de $f$ será
  $D= \{ x \in \R | x \geq 0 \}$,
  uma vez que o número de quilômetros rodados é sempre não-negativo.
\end{observation}

Note que o contradomínio $C$ de uma função $f$ pode ser qualquer conjunto que
contenha, dentre outros elementos, os valores de $f(x)$. Sendo assim, em muitas
situações estamos interessados em trabalhar apenas com o conjunto que contém
apenas os valores gerados por $f$, ao qual damos o nome de \emph{conjunto
imagem}.

\begin{definicao}{Imagem}{def:imagem}
  Dada uma \emph{função} $f:D\to C$, o valor $f(x)$ para $x\in D$ é chamado
  \emph{imagem de $x$}. O conjunto de todas as imagens da função $f$ é
  denominado \emph{conjunto imagem}(ou simplesmente imagem) e denotado por
  \emph{\emph{Im}(f)}. Ou seja,  \emph{Im}(f)$=\{f(x);x \in D\}$.
\end{definicao}

\begin{definicao}{}{def:} (\emph{Gráfico})
  o \emph{gráfico} de $f$ é o conjunto dos pontos $(x,y)$ do plano cartesiano
  que satisfazem $y=f(x),$ ou seja,  $G(f)=\{(x, f(x)); x\in D\}$.
\end{definicao}

\emph{Quando que uma aplicação não é função?}

\begin{example}
  Verifique se a equação $y^2+x^2-16=0$ permite uma definição
  de $y$ como função de $x$.

  \solution
  Inicialmente vamos ver se conseguimos explicitar $y$ como função de $x$.
  \begin{align*}
  y^2 & =    16-x^2 \\
  y   & =\pm \sqrt{16-x^2}
  \end{align*}
  Logo, para cada $x \in \mathbb{R}$, temos dois valores para $y$. Portanto, a
  equação $y^2+x^2-16=0$ não permite uma definição de $y$ como função de $x$.
\end{example}

\begin{example}
  Faça o estudo do domínio das seguintes funções
  \begin{mathlist}
  \item f(x)=1-x^4
  \item g(x)=\frac{1}{x+4}
  \item h(x)=\sqrt{9-x}
  \item f(t)=\frac{\sqrt{2+t}}{\sqrt{2t-6}}
  \item f(x)=\sqrt[3]{2x-3}
  \item p(x)=\frac{\sqrt{x-1}}{x^2-5x+6}
  \end{mathlist}

  \solution
  \emph{a)} Note que $f$ está bem definida para todo $x \in \mathbb{R}$, logo
  $D(f)= \mathbb{R}$.

  \emph{b)} Nesse caso temos que $g$ está definida quando $x+4 \neq 0
  \Rightarrow x \neq -4$, logo $D(g)= \mathbb{R}-\{-4\}$ ou $D(g)= \{ x \in
  \mathbb{R}| x \neq -4 \}$.

  \emph{c)} Para que $h$ seja bem definida $9-x \geq 0 \Rightarrow  x \leq 9
  $, logo $D(h)= \{ x \in \mathbb{R}| x \leq 9 \}$.

  \emph{d)} Para que $f$ esteja bem definida teremos que ter $2+t \geq 0
  \Rightarrow t \geq -2$ e $2t-6 >0 \Rightarrow t>3$, fazendo a interseção
  desses intervalos temos que $D(f)= \{ x \in \mathbb{R}| x \geq 3 \}$.

  \emph{e)} Como o índice a raíz é ímpar, então $D(f)= \mathbb{R}$.

  \emph{f)} Para que $p$ esteja bem definida teremos que ter
  $x-1 \geq 0 \Rightarrow x \geq 0$
  e
  $x^2-5x+6 \neq 0 \Rightarrow x \neq 2$
  e
  $x \neq 3$,
  portanto
  $D(p)= \{ x \in \R | x \geq 1, x \neq 2 \hspace{0,15cm} x \neq 3 \}$.
\end{example}

\subsection{Simetria}

Se uma função $f$ satisfaz $f(-x) = f(x)$ para todo número $x$ em seu
domínio, então $f$ é chamada \emph{função par}. Por outro lado, se $f$
satisfaz $f(-x) = -f(x)$ para cada número $x$ em seu domínio, uma $f$  é
\emph{função ímpar}.

\begin{example}
  Verifique se as funções abaixo são par, ímpar ou nem par, nem ímpar \\
  a) $f(x)=1-x^4$  \hspace{1cm}
  b) $g(x)=x^5+x$  \hspace{1cm}
  c) $h(x)=2x-x^2$ \\

  \emph{a)} $f(-x)=1-(-x)^4=1+x^4=f(x)$, logo $f$ é par. \\
  \emph{b)} $g(-x)=(-x)^5-(-x)= -x^5+x=-(x^5-x)=-g(x)$, logo $g$ é ímpar. \\
  \emph{c)} Nesse caso temos que $h(1)=2.1-1^2=1$ e $h(-1)=2(-1)-(-1)^2==-3$,
  logo $h$ não é nem par, nem ímpar.
\end{example}

\subsection{Zeros de função}

Os valores de $x$ que satisfazem a equação $f(x)=0$ são chamados \emph{zeros}
de $f$. Esses valores correspondem aos interceptos-x do gráfico da função.

\begin{example}
  Encontre os zeros das seguintes funções:
  a) $f(x)=x^3+2x^2-5x$
  b) $g(x)= \frac{x^2-4}{x+3}$ \\

  \emph{a)} Para encontrarmos os zeros de $f$ precisamos resolver a equação
  $x^3+2x^2-5x=0 \Rightarrow x(x^2+2x-5)=0$, portanto $x=0$ ou $x^2+2x-5=0$.
  Para encontrarmos as raízes de $x^2+2x-5=0$ usamos a fórmula de Bhaskara, uma
  vez que se trata de uma equação do segundo grau. Com isso, $x^2+2x-5=0
  \Rightarrow x=-1+\sqrt{6}$ ou $x=-1-\sqrt{6}$. Logo, $f$ possui três zeros, a
  saber $x=0$, $x=-1+\sqrt{6}$ e $-1-\sqrt{6}$. \\

  \emph{b)} Para encontrarmos os zeros de $g$ precisamos resolver a equação
  $\frac{x^2-4}{x+3}=0 \Rightarrow x^2-4=0 \Rightarrow x=2 \hspace{0,3cm} ou
  \hspace{0,3cm} x=-2$. Logo $g$ possui dois zeros, a saber $x=-2$ e $x=2$.
\end{example}

\subsection{Monotonicidade}

\begin{definicao}{}{def:}
  Seja $f$ uma função definida em um intervalo $I$, dizemos que
  \begin{enumerate}
  \item $f$ é \emph{crescente} em $I$ se, dados quaisquer $x_1$ e $x_2$ em
    $I$, tais que $x_1< x_2$, tivermos $f(x_1)< f(x_2)$;

  \item $f$ é \emph{decrescente} em $I$ se, dados quaisquer $x_1$ e $x_2$
    em $I$, tais que $x_1< x_2$, tivermos $f(x_1)> f(x_2)$;

  \item $f$ é \emph{constante} em $I$ se, dados quaisquer $x_1$ e $x_2$ em
    $I$, tivermos $f(x_1)=f(x_2)$

  \end{enumerate}
\end{definicao}

\subsection{Função Injetora}

\begin{definicao}{}{def:}(\emph{Função Injetora}):
  Uma função $f$, definida em seu domínio $D$ é \emph{injetora} quando dados
  quaisquer valores reais $x_1$, $x_2 \in D$, se $x_1 \neq x_2$, então, $f(x_1)
  \neq f(x_2)$.
\end{definicao}

Em outras palavras, se $f$ é uma função injetora então,
\[
  f(x_1)=f(x_2) \Leftrightarrow x_1=x_2.
\]

\emph{Teste da reta horizontal:}
Uma função é injetora se e somente se nenhuma reta
horizontal intercepta seu gráfico mais de uma vez.

\begin{figure}
  \centering
  \includegraphics[scale=1.4]{injet}
  \caption{Gráfico de uma função que não é injetora,
  uma vez que $f(x_1)=f(x_2)$ para $x_1 \neq x_2$.}
\end{figure}

\begin{example}
  Verifique se as funções abaixo são injetoras \\
  a) $f(x)=x^2$ \hspace{1cm}
  b) $f(x)=x^3$ \hspace{1cm}
  c) $y= \frac{3}{5x-2}$ \\

  \emph{a)} $f$ não é injetora pois $f(-1)=f(1)=1$. \\

  \emph{b)} Suponha que $f(x_1)=f(x_2) \Rightarrow x_1^3=x_2^3 \Rightarrow
  x_1^3= \sqrt[3]{x_2^3} \Rightarrow x_1=x_2$, logo $f$ é injetora. \\

  \emph{c)} Suponha que $f(x_1)=f(x_2) \Rightarrow \frac{3}{5x_1-2}=
  \frac{3}{5x_2-2} \Rightarrow 3(5x_1-2)=3(5x_2-2) \Rightarrow 5x_1-2=5x_2-2
  \Rightarrow x_1=x_2.$ Logo, $y$ é injetora.
\end{example}

\subsection{Algumas funções usuais}

A seguir serão apresentados algumas funções usuais, onde algumas delas serão
estudadas com mais detalhes em aulas posteriores.

\emph{Exemplo 1 (Função Potência):} A função $f(x)=x^ n$, onde $n$ é um
número natural, é chamada de \emph{função potência}. A figura  \ref{pot}
mostra o gráfico de algumas funções potências.

\begin{figure} \label{pot}
  \centering
  \includegraphics[scale=1.6]{pot1}
  \includegraphics[scale=1.6]{pot2}
\end{figure}

\begin{itemize}
  \item O domínio de $f$ é $\mathbb{R}$.

  \item O conjunto imagem é $\mathbb{R}$ se o expoente for ímpar e é
    $[0,\infty]$ se o expoente é par.

  \item Há um único zero em $x=0$.

  \item Quando o expoente é par, $f$ é par, quando o expoente é ímpar, $f$ é
    ímpar.

  \item Quando o expoente é par, f é decrescente para $x<0$, e crescente para
    $x>0$, quando o expoente é ímpar, $f$ é crescente em $(- \infty, \infty)$.

\end{itemize}

\emph{Exemplo 2 (Função Polinomial):} A função $f(x)=a_nx^ n+
a_{n-1}x^{n-1}+...+ a_2x^2+a_1x+a_0$. $n$ é um número inteiro não- negativo e
$a_1,a_2,...a_n$ são constantes reais.

\begin{itemize}
  \item O domínio de $f$ é $\mathbb{R}$.
  \item Pode ter até $n$ zeros.
\end{itemize}
Os exemplos 3 e 4 ilustram exemplos de funções polinomiais.

\emph{Exemplo 3: (Função do 1\textsubscript{o} grau)}
É uma função do tipo $f(x)=ax+b$, com $a,b \in \mathbb{R}$. O gráfico de uma
função do 1\textsuperscript{o} grau é uma reta. $a$ é o coeficiente angular da reta e
$b$ é a intersecção com o eixo $y$. Para encontrar a equação de uma reta
precisamos de dois pontos, em particular se encontrarmos os interceptos com os
eixos conseguimos obter a equação da reta.

\emph{Exemplo 5: (Função raíz)} É uma função do tipo $f(x)=\sqrt[n]{x}$, com
$n$ sendo uma constante natural maior que $1$.

\begin{itemize}
 \item O domínio é $\mathbb{R}$ se o índice $n$ é ímpar e é $[0,\infty)$ se o
   índice é par.

 \item O conjunto imagem é $\mathbb{R}$ se o índice $n$ é ímpar e é
   $[0,\infty)$ se o índice é par.

 \item Há um único zero em $x=0$.

 \item A função é crescente em todo o domínio.

 \item Quando o índice é ímpar, $f$ é ímpar.
\end{itemize}

\emph{Exemplo 6: (Função recíproca)}
É uma função do tipo $f(x)=\frac{1}{x^n}$, com $x \neq 0$.
\begin{itemize}
  \item o domínio inclui todos os valores reais, exceto o zero.

  \item o conjunto imagem é $(-\infty,0) \cup (0, \infty)$ se o expoente é
    ímpar e $(0, \infty)$ se o expoente é par.

  \item A função não tem zeros.

  \item Quando o expoente é ímpar, a função é decrescente em todos os pontos do
    domínio. Já quando o expoente é par, $f$ é crescente para $x<0$ e
    decrescente para $x>0$.

  \item Quando o expoente é par, $f$ é par. Quando o expoente é ímpar, $f$ é
    ímpar.

\end{itemize}

\begin{figure}
  \centering
  \includegraphics[scale=0.7]{hiperbole}
  \caption{Gráfico da função $\frac{1}{x}$.}
\end{figure}

\emph{Exemplo 7: (Função racional)}  É uma função do tipo
$f(x)=\frac{p(x)}{q(x)}$, onde $p(x),q(x)$ são polinômios e $q(x) \neq 0$. Por
exemplo, a função $\frac{2x^4-x^2+1}{x^2-4}$ é uma função racional e seu
domínio é $\{ x \in \mathbb{R} | x \neq \pm 2 \}$.

\emph{Exemplo 8: (Função exponencial)} É uma função do tipo $f(x)=a^x$. onde
$a$ é uma constante positiva.

\begin{itemize}
  \item O domínio de $f$ é $\mathbb{R}$ (ou $(- \infty, \infty)$).

  \item A imagem é $(0, \infty)$.

  \item A função não possui zeros;

  \item se $a>1$, a função é crescente em todo o domínio e se $0<a<1$ a função
    é decrescente em todo o domínio.

  \item A função não é par nem ímpar.

\end{itemize}

\emph{Exemplo 9: (funções trigonométricas)} As funções trigonométricas são
funções angulares obtidas através do auxílio do círculo trigonométrico. As
principais funções trigonométricas são as funções $y= \sin(x)$ e $y=\cos(x)$ e
$y=tg(x)$. Essas funções são períodicas, ou seja,  os valores da função $(f(x)
= y)$ se repetem para determinados valores da variável $x$, ou seja, para cada
período determinado pelos valores de $ x$, iremos obter valores repetidos para
a função.

\begin{itemize}
  \item O domínio de ambas funções $y=\sin(x)$ e $y=\cos(x)$ é $\mathbb{R}$.

  \item A Imagem de ambas funções $y=\sin(x)$ e $y=\cos(x)$ é $[-1,1]$, ou
    seja, $-1 \leq \sin(x) \leq 1$ e $-1 \leq \cos(x) \leq 1$.

  \item A função $y=\sin(x)$ é ímpar e a função $y= \cos(x)$ é par.

\end{itemize}

\emph{Exemplo 10:(Funções definidas por partes)}: Uma função definida por
partes é uma função que tem mais de uma lei para o seu domínio.
A função
\[
  f(x)=\left\{\begin{array}{cc}
    x^2-x \,\, se \,\, x < 1\\
    x-1 \,\, se \,\, x \geq 1
  \end{array}\right.
\]
é um exemplo de função definida por partes.

\subsection{Função composta}

\begin{definicao}{}{def:} \label{com}\emph{Função composta}]
  Dadas as funções $f $ e $ g$ cujos domínios são $A$ e $B$, respectivamente,
  definimos a \emph{função composta} $f \circ g$ por
  \[
    (f \circ g)(x)= f(g(x))
  \]
  O domínio de $f \circ g$ é o conjunto dos valores de $x \in B$ tais que $g(x)
  \in A$
\end{definicao}

\begin{example}
  Se $f(x)= x^2$ e $g(x)=x-3$, encontre as compostas
  $f \circ g$ e $g \circ f$. \\
  Por definição
  $f \circ g= f(g(x))=f(x-3)=(x-3)^2$ e
  $g \circ f= g(f(x))=g(x^2)= x^2-3$.
\end{example}

\begin{example}
  Se $f(x)= \frac{x}{x+1}$, $g(x)=x^{10}$ e $h(x)=x+3$,
  encontre  $f \circ g \circ h$.
  \[
    f \circ g \circ h
    = f(g(h(x)))
    = f(g(x+3))
    =f((x+3)^{10})
    =\frac{(x+3)^{10}}{(x+3)^{10}+1}.
  \]
\end{example}

\begin{observation} \label{observation}
  Devemos está atentos ao calcularmos o domínio da função composta. Segundo a
  definição \ref{com} para que sejamos capazes de calcular $f(g(x))$ é preciso
  que:
  \begin{itemize}
    \item[1.] $x$ pertença ao domínio de $g$;
    \item[2.] $g(x)$ (o valor de $g$ em $x$) pertença ao domínio de $f$;
  \end{itemize}
  Para deixar mais clara a definição do domínio da função composta, veremos
  como obtê-lo de forma prática no exemplo a seguir.
\end{observation}

\begin{example}
  Sejam dadas as funções $f(x)=\frac{1}{x^2-4}$ e $g(x)=\frac{1}{x}$. Obtenha
  $f(g(x))$ e determine o domínio dessa composta. \\

  $f(g(x))= f(\frac{1}{x})=\frac{1}{\frac{1}{x^2}-4}=
  \frac{1}{\frac{1-4x^2}{x^2}}= \frac{x^2}{1-4x^2}$. Pela Observação  \ref{observation}
  para que $f(g(x))$ esteja bem definida temos que ter : \\

  \emph{1.} $x$ pertença ao domínio de $g$, isto é, $x \neq 0$. \\

  \emph{2.} $g(x)$ pertença ao domínio de $f$, isto é $1-4x^2 \neq 0
  \Rightarrow x \neq -\frac{1}{2}$ e $x \neq \frac{1}{2}$ . \\

  Logo  $D(f(g(x)))= \{ x \in \mathbb{R} | x \neq 0, x \neq -\frac{1}{2}, x
  \neq \frac{1}{2}\}$
\end{example}
