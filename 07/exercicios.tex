\section{Exercícios}

\begin{enumerate}
\item A polícia federal localizou na floresta amazônica uma pista de pouso clandestina com as
seguintes características:
\begin{itemize}
\item A pista media 300 metros de comprimento, era plana e horizontal;
\item No final da pista havia uma árvore de 30 m de altura.
\end{itemize}
Um pequeno avião encontra-se em um ponto A da pista a 300 metros da árvore. O avião partirá desse ponto A, em direção à árvore e levantará vôo após percorrer 180 metros de pista. Se o avião levantar vôo em linha reta, de modo que essa reta (trajetória do avião no ar) forme um ângulo $\alpha$ com o plano horizontal, qual deverá ser a tangente de $\alpha$ para que a aeronave passe exatamente a 10 metros acima da árvore?

\item Uma TV encontra-se fixada em uma parede e a dois metros de distância dessa parede está posicionado um sofá de três lugares. Uma pessoa senta-se em um dos lugares das extremidades dos sofá de forma que sua distância até a TV é de 3 metros. O ângulo de visão dessa pessoa é o ângulo formado entre a reta que liga a visão da pessoa ao centro da TV e a reta perpendicular à parede passando pelo centro da TV. Sabendo que a visão da TV só é boa se a pessoa assiste TV em um ângulo menor ou igual a $\pi/3$, verifique se essa pessoa terá uma boa visão da TV.
\item Dê os valores dos ângulos a seguir e radianos e calcule o valor das funções seno, cosseno, tangente, secante, cossecante e cotangente nos mesmos:
\begin{enumerate}
  \item $150^{\circ}$
  \item $315^{\circ}$
  \item $210^{\circ}$
\end{enumerate}

\item Sabendo que $\sin(x)=-\frac{3}{4}$ e $\pi \leq x \leq \frac{3\pi}{2}$, determine o valor de $\cos(x)$, $\tan(x)$ e $\sec(x)$

 \item Sabendo que $\sin(\theta)=\frac{1}{4}$ e que $\theta$ está no segundo quadrante, determine o valor de $\cos(\theta)$, $\tan(\theta)$ e $\sec(\theta)$.
 \item Sabendo que $\cos(x)= \frac{3}{5}$, determine os possíveis valores de $\sin(x)$ e os quadrantes aos quais $x$ pode pertencer.
\item Trace os gráficos das funções a seguir incluindo ao menos um período, indique também qual a imagem, o valor máximo e o valor mínimo de cada função:
\begin{enumerate}
\item $f(x)=\tan(x+\pi/4)$
\item $f(x)=\cos(4x)$
\item $f(x)=1-2\sin (x)$
\item $f(x)=\sin(x)+2$
\end{enumerate}

    \item[b)] $\frac{\sin(4x)}{\sin(x)\cos(2x)}$
    \item[c)] $\cot x \hspace{0,1cm} \sec x \hspace{0,1cm} \csc x $
\end{itemize}
 \item Faça a redução ao  primeiro quadrante e encontre o valor de:

         \item $\cos\left( \frac{37\pi}{6}\right)- \sin(6\pi)$
         \item $\tan(315^{\circ})-\cos(330^{\circ})$
     \end{enumerate}

\item  Qual o maior e o menor valor que $f(x) = 5+5\sin(5x+5)$ assume?

    \item Determine quanto tempo a roda gigante demora para dar uma volta completa (ou seja, o período de $h(t)$).
    \item Usando um programa gráfico, trace a curva $h(t)$.
\end{enumerate}
\item Usando a fórmula de adição de arcos, encontre os valores de $\sin(75^{ \circ})$, $\cos(75^{ \circ})$ e $\tan(75^{ \circ})$. Dica: Note que $75^{\circ}=30^{\circ}+ 45^{\circ}$.
 \item Nesse exercício você poderá usar as seguintes relações trigonométricas:
   \[\sin(x+y)= \sin(x)\cos(y)+\sin(y)\cos(x) \,\, e \,\, \cos(x+y)=\cos(x)\cos(y)-\sin(x)\sin(y)\]
    \begin{enumerate}

        \item Determine $\sin(x-y)$, $\cos(x-y)$ e $\tan(x-y)$, sabendo que $\sin(x)=\frac{4}{5}$, $\cos(y)=\frac{\sqrt{5}}{5}$ e que $x,y \in 1^{\circ}$ quadrante.
        \item  Encontre o valor de $\sin(15^{\circ})$ e a partir desse resultado, encontre o valor de $\sin(7,5^{\circ})$.
        \item Mostre que $\cos \left(x + \frac{\pi}{2}\right)=-\sin(x)$.
        \item Encontre o valor da expressão $\ln(\sec(\theta))+\ln(\cos(\theta))$
        \item Econtre o valor da espressão
          \[\cos(\arctan(x)+\arcsin(\frac{1}{\sqrt{3}}))\]
        obs: Considere que os ângulos envolvidos nesse item estão sempre no primeiro quadrante.
    \end{enumerate}
    \item Sabe-se que $x + y = \frac{\pi}{4}$ e $\sin(x) = \frac{5}{13}$, com $0 \leq x \leq \frac{\pi}{2}$, calcule $\sin(y)$ e $\cos(y)$.

\item A partir do gráfico de $f(x)= \cos(x)$, faça o esboço do gráfico de $g(x)=-1+ 2 \cos(x+ \frac{\pi}{2})$. Encontre também o período e a imagem.


\item Resolva as seguintes equações trigonométricas, no intervalo especificado:
\begin{enumerate}
 \item Encontre os valores de $x$ no intervalo $[0,2\pi]$ para os quais $2\sin(x)=-1$.
 \item Qual é a soma das duas menores soluções positivas de $\sin\left(3x-\frac{\pi}{4}\right)=0$?
 \item Encontre os valores de $x\in\R$ para os quais $\sin(2x)=\cos x$.
 \item $2 \sin^2(x)- \sin(x)=0$, no intervalo $[0,2\pi]$.
 \item $2\sin^2(x)+\cos(x)=1$, no intervalo $[0,2\pi]$.
 \item $\tan^2(\theta) -3 = 0,$ no intervalo $[-\frac{\pi}{2},\frac{\pi}{2}]$.
\end{enumerate}
\item Encontre
\begin{enumerate}
    \item $\arcsin(\frac{\sqrt{3}}{2})$
    \item $\tan(\arccos(\frac{\sqrt{5}}{4}))$
    \item $\arctan(-\sqrt{3})$
    \item $\tan(\arctan(5))$
\end{enumerate}
\item Calcule o valor da expressão $\arcsin(-1)+\arccos(\frac{1}{2})-\arctan(\tan(\frac{\pi}{4}))$
\item Determine o domínio das seguintes funções
\begin{enumerate}
    \item $f(x)=\arcsin(x-3)$
    \item $g(x)=\arccos(x^2-1)$
\end{enumerate}
\item Use a Relação fundamental trigonométrica para mostrar que $\cos(\arcsin(x))=\sqrt{1-x^2}$, para $x \in [-1,1]$.
\end{enumerate}
