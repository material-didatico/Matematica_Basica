\subsection{Exercícios}

\begin{enumerate}

  \item Faça o esboço gráfico das funções abaixo

  \begin{enumerate}
    \item $f(x) = 2^x-2$
    \item $f(x)=\left( \frac{1}{5}\right)^x$
    \item $f(x) =-4\left(\frac{1}{2}\right)^x$
    \item $f(x) = 3^{-x}$
  \end{enumerate}
   \item  Encontre o domínio das seguiuintes funções:
   \begin{enumerate}
   \item  $g(x)=\frac{-5}{ \sqrt{16-\left( \frac{1}{2} \right)^x}}$
       \item  $f(x)= \sqrt{9. 3^{2x-1}-\frac{1}{27}}$.
   \end{enumerate}

  \item Resolva as equações abaixo
  \begin{enumerate}
    \item $8^{x}=0,25$
       \item $5^{2x-7}=125$
    \item $8^{x^2-x}=4^{x+1}$
    \item $2^{x^2- 3x - 4}=1$
  \item $\sqrt{\left(\frac{1}{2}\right)^{3x-2}}=\left(\frac{1}{2}\right)^{-4x}.2^{-x+4}$
    \item $(10^x)^{1-x}-0,000001=0$
  \item $2^{2x + 1} - 3.2^{x + 2} = 32$
         \item $2^{x^2+5}+2^4=144$
         \item $5^{4x-1}-5^{4x}-5^{4x+1}+5^{4x+2}=480$
        % \item $e^{2x}+2e^x-8=0$ \textbf{Gabarito:} $S=\{\ln(2)\}$
         \item $2^{3x-2}-4^{x+6}=0$
      \end{enumerate}
    \item Resolva as inequações abaixo
    \begin{enumerate}
      \item $\left(\frac{1}{2}\right)^{2x-1}\leq \frac{1}{16}$
      \item $e^{x^2-5}\geq e^{4x}$
       \item $\left(\frac{1}{2} \right)^{2x-1}> \frac{1}{16}$
         \item $7^{x-2}<343$
         \item  $1 \leq \left( \frac{1}{2}\right)^x \leq 8$
    \end{enumerate}


    \item Se $f(x) = 16^{1+1/x}$ , encontre o valor de $f(-1) + f(-2) + f(-4)$

    \item Calcule $f(0)-f (3/2)$ sabendo que $f(x)=\begin{cases}
    2^x,&\text{se }-1\leq x\leq 1,\\
    1/x,&\text{se }x>1.
    \end{cases}$

    \item Seja a função $f:\mathbb{R}\to\mathbb{R}$ definida por $f(x) = 2^x$ . Encontre o valor de $f(a+1)-f(a)$

\end{enumerate}
