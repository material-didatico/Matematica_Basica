\begin{exercise}
  Um clube vem perdendo sócios a um ritmo constante ano após ano, tendo
  registrado 14 mil sócios em 2004 e apenas 8 mil sócios em 2013.
  \begin{alphalist}
    \item Defina uma função linear (ou afim) $S(t)$ que forneça o número
          de sócios do clube no ano $t$, supondo que $t$ seja zero no
          ano 2000.
    \item O clube pretende fechar as portas se o número de sócios chegar a
          2 mil. Usando a função encontrada no item a),
          determine em que ano isso ocorrerá.
    \item Construa o gráfico de $S$.
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item \(S(t) = -\frac{2}{3}t + \frac{50}{3}\)
      \item 2022
      \item \IncludeGraphicCentred{gabarito_aula2-2c}[0.9]
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  (Minimização da área cercada) Um fazendeiro pretende usar \SI{500}{\meter}
  de cerca para proteger um bosque retangular de comprimento $x$
  e largura $y$, às margens de um riacho.
  \begin{alphalist}
    \item Usando o comprimento da cerca, escreva o valor de $y$ em função de $x$.
    \item Com base na expressão encontrada no item a), escreva a função
          $A(x)$ que fornece a área cercada com relação a $x$.
    \item Determine o valor de $x$ que maximiza a área cercada.
          Determine também o valor de $y$ e a área máxima.
    \item Trace o gráfico de $A(x)$.
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $y = 250-x$
      \item $A(x) = 250x - x^2$
      \item $x=y=\SI{125}{\meter}$ e $A_{\text{max}} = \SI{15625}{\meter^2}$
      \item \IncludeGraphicCentred{gabarito_aula2-4d}[0.9]
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Dada a função $f(x)= x^2-3x$,
  \begin{alphalist}
    \item Determine algebricamente os pontos no quais $f(x)=0$.
    \item Determine algebricamente os pontos no quais $f(x)=-2$.
    \item Esboce o gráfico da função no plano coordenado, indicando os
          pontos que foram obtidos no item anterior.
    \item Determine graficamente a solução da inequação $f(x) \geq -2$.
          Confirme sua resposta resolvendo analiticamente a inequação.
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $x=0$ e $x=3$
      \item $x=1$ e $x=2$
      \item \IncludeGraphicCentred{gabarito_aula2-5c}[0.9]
      \item $x \leq 1$ ou $x \geq 2$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Simplifique as expressões, fatorando os termos, caso necessário.
  Suponha sempre que os denominadores são não nulos.
  \begin{mathlist}
    \item \frac{2x-6}{x-3}
    \item \frac{x^2-x^3}{x}
    \item \frac{x^2-3x}{4x-12}
    \item \frac{2x^2-50}{x^3+5x^2}
    \item \frac{x^3-8}{x-2}
    \item \frac{x^2y-xy^2}{xy}
    \item \frac{x^2-9}{x^2-3x}
    \item \frac{x^4-y^4}{x-y}
    \item \frac{x^2-5x+4}{x-4}
    \item \frac{x-9}{\sqrt{x}+3}
  \end{mathlist}

  \begin{answer}
    \begin{alphalist}
      \item \(2\), se $x \neq 3$
      \item \(x(1-x)\), se $x \neq 0$
      \item \(\frac{x}{4}\), se $x \neq 3$
      \item \(\frac{2(x-5)}{x^2}\), se $x \neq 5$
      \item \(x^2+2x+4\), se $x \neq 2$
      \item \(x-y\), se $xy \neq 0$
      \item \(\frac{x+3}{x}\), se $x \neq 3$
      \item \((x+y)(x^2+y^2)\), se $x-y \neq 0$
      \item \(x-1\), se $x \neq 4$
      \item \(\sqrt{x}-3\)
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Encontre as raízes dos polinômios a seguir e escreva-os na forma fatorada.
  \begin{mathlist}
    \item p(x) = 9x^4-10x^2+1
    \item q(x) = x^3-4x^2-17x+60
    \item g(x) = 2x^3-3x^2-23x+12
    \item f(x) = x^4-4x^3+5x^2-2x
  \end{mathlist}

  \begin{answer}
    \begin{alphalist}
      \item Raízes: \(x=\pm 1\), \(x=\pm \frac{1}{3}\).
            Fatoração: \(p(x)=9(x+1)(x-1)(x-\frac{1}{3})(x+\frac{1}{3})\)
      \item Raízes: $x=-4, 3, 5$.
            Fatoração: \(q(x)=(x+4)(x-3)(x-5)\)
      \item Raízes: \(x=-3, \frac{1}{2}, 4\).
            Fatoração: \(p(x)=2(x+3)(x-\frac{1}{2})(x-4)\)
      \item Raízes: $x=0, 1, 2$.
            Fatoração: \(p(x)=x(x-1)^2(x-2)\)
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}

  \begin{alphalist}

    \item $g(x)=6x^2+7x+9$    por $2x^2-5x+1$

      \item $Q(x)=2x^3+ 3x^2+2x+\frac{2}{3}$, $R(x)=\frac{4}{3}$
      \item $Q(x)=3$, $R(x)=22x+6$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Resolva as inequações
  \begin{mathlist}
    \item 3x-5 \leq 0
    \item (x+2)(x-3) \leq 0
    \item x^2-3x \geq 0
    \item -4x^2+12x-9 \leq 0
    \item \frac{x-2}{x+3} \leq 0
    \item \frac{x+8}{x^2+7x+12} \leq 0
  \end{mathlist}

  \begin{answer}
    \begin{alphalist}
      \item $S=\{ x \in \R \st  x \leq \frac{5}{3} \}$
      \item $S=[-2,3]$
      \item $S=\{ x \in \R \st  x \leq 0 \text{ ou } x \geq 3 \}$
      \item $S= \R$
      \item $S=\{ x \in \R \st -3< x \leq 2 \}$
      \item $S=\{ x \in \R \st  x \leq -8 \text{ ou } -4<x<-3 \}$
    \end{alphalist}
  \end{answer}
\end{exercise}
