\begin{exercise}
  Uma companhia de turismo cobra \reais{1200} por um pacote turístico
  individual, mas dá um bom desconto quando os turistas viajam em grupo. Se um
  grupo tem $x$ pessoas, com $x \leq 70$, cada integrante paga apenas
  \[
    C(x) = 1200 - 12x.
  \]
  Sabendo que a receita da empresa em cada grupo é dada pelo produto do valor
  pago por pessoa pelo número de membros do grupo.
  \begin{alphalist}
    \item escreva a função $R(x)$ que fornece a receita obtida pela empresa de
          turismo com o grupo de $x$ pessoas
    \item trace o gráfico da função para $x \in [10, 70]$
    \item indique qual o tamanho do grupo fornece a maior receita
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $R(x)=x(1200-12x)$
      \item \IncludeGraphicCentred{gabarito_aula1-1b}[0.9]
      \item $x=50$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Calcule
  \[
    \frac{f(x)-f(a)}{x-a}
  \]
  para as funções e os valores de $a$ fornecidos a seguir. Simplifique os
  resultados e suponha sempre que os denominadores são diferentes de zero.
  \begin{mathlist}
    \item f(x) = 3x+4,  \qquad a=2
    \item f(x) = x^2+6, \qquad a=4
  \end{mathlist}

  \begin{answer}
    \begin{alphalist}
      \item 3,     se $x \neq 2$
      \item $x+4$, se $x \neq 4$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  O gráfico da função
  \[
    f(x)= \sqrt{-x^2+2x+3}
  \]
  é mostrado a seguir.
  \IncludeGraphicCentred{questao_aula1-3}[0.7]
  Determine:
  \begin{alphalist}
    \item Os valores de $f(0)$, $f(0,5)$ e $f(2)$
    \item O domínio de $f$
    \item O conjunto imagem de $f$
    \item Os zeros de $f$
    \item Os intervalos de crescimento e decrescimento
    \item Os pontos de máximo e mínimo absoluto
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $f(0)=f(2)= \sqrt{3}$ e $f(0,5)=\frac{\sqrt{15}}{2}$
      \item $D(f)= \{ x \in \R | -1 \leq x \leq 3\}$
      \item Note que o valor mínimo de $f$ é $0$ e o valor máximo é $2$, assim,
            $\Im(f)= [0,2]$
      \item $x=-1$ e $x=3$
      \item Pelo gráfico podemos notar que $f$ é crescente no intervalo $[-1, 1]$
            e decrescente no intervalo $[1,3]$
      \item $(-1,0)$ e $(3,0)$ são os pontos de mínimo absolutos e $(1,2)$
            é o ponto de máximo absoluto
    \end{alphalist}
  \end{answer}
\end{exercise}

  \begin{mathlist}
    \item f(x) = \frac{1}{x}+ \frac{1}{x+5}
    \item f(x) = \sqrt{x+2}+ \sqrt{x-5}

  \begin{answer}

      \item D(f) = \left\{ x \in \R \st x \geq \frac{3}{2},\, x \neq 4 \right\}
      \item D(h) = \left\{ x \in \R \st x < 0 \text{ ou } x > 5        \right\}
    \end{mathlist}
  \end{answer}

\begin{exercise}
  Se $f(x)=x+2$, $g(x)=x^2-4$ e $h(x)=-2x+5$
  \begin{alphalist}
    \item Verifique se cada uma dessas funções são par, ímpar ou nem par nem ímpar
    \item Verifique se essas funções são injetoras
    \item Esboce o gráfico dessas funções, encontre o conjunto imagem e
          os intervalos de crescimento e decrescimento
    \item Encontre o valor de \(\frac{g(1)-f(0)-h(3)}{h(1)}\)
    \item Calcule $g(h(2))$
    \item A partir do gráfico de $y=2^x$, esboce o gráfico de $y= 2^{f(x)}$
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $f$ é nem par, nem ímpar, $g$ é par e $h$ é nem par, nem ímpar
      \item $f$ e $h$ são injetoras e $g$ não é injetora
      \item $f$ é sempre crescente, $g$ é decrescente no intervalo $(-\infty, 0)$
            e crescente no intervalo $(0, \infty)$ e $h$ é sempre decrescente
            \IncludeGraphicCentred{gabarito_aula1-6cI}[0.9]
            \IncludeGraphicCentred{gabarito_aula1-6cII}[0.9]
            \IncludeGraphicCentred{gabarito_aula1-6cIII}[0.9]
      \item $-\frac{4}{3}$
      \item $g(h(2))=-3$
      \item \IncludeGraphicCentred{gabarito_aula1-6f}[0.9]
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  (\textbf{Cercando um pasto}) Um fazendeiro pretende usar \SI{400}{\meter}
  de cerca para delimitar uma área retangular que servirá de pasto.
  Responda os itens a seguir, lembrando que a área de um terreno retangular
  com largura $l$ e profundidade $p$ é dada por $lp$,
  e que o perímetro desse terreno retangular é igual a $2l+2p$.
  \begin{alphalist}
    \item Relacione a profundidade com a largura do pasto, considerando o uso
          dos \SI{400}{\meter} de cerca.
    \item Escreva uma função que forneça a área cercada em relação a largura
          do pasto.
    \item Calcule a área do pasto, supondo que sua largura é \SI{75}{\meter}.
          Faça o mesmo para uma largura de \SI{150}{\meter}.
    \item Determine o domínio da função área que você obteve.
    \item Esboce o gráfico da função área.
    \item Indique os intervalos nos quais a função é crescente e em quais
          ela é decrescente.
    \item Com base no gráfico, indique se é possível cercar uma área de
          \SI{12000}{\meter^2}.
    \item Ainda com base no gráfico, determine a maior área que pode ser
          cercada e as dimensões do terreno nesse caso.
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $p=200-l$
      \item $A(l)=200l-l^2$
      \item $A(75)=9375 m^2$ e $A(150)=7500m^2$
      \item $D=(0,200)$, para chegar nesse domínio você deve impor que $A(l)> 0$
      \item \IncludeGraphicCentred{gabarito_aula1-8e}[0.9]
      \item É crescente no intervalo $(0,100) $ e decrescente no intervalo $(100,200)$
      \item Não, pois a equação $200l-l^2=12000$ não possui solução real
      \item $A_{max}=10000m^2$ e as dimensões são $l=p=100$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Dadas as funções $f$ e $g$ a seguir,
  defina $f(g(x))$, $g(f(x))$, $f(f(x))$ e $g(g(x))$.
  \begin{mathlist}
    \item f(x) = 4x,       \quad g(x) = \frac{x^2}{4}
    \item f(x) = \sqrt{x}, \quad g(x) = \frac{x}{3}
  \end{mathlist}

  \begin{answer}
    \begin{alphalist}
      \item $f(g(x))=x^2$,
            $g(f(x))=4x^2$,
            $f(f(x))=16x$ e
            $g(g(x))=\frac{x^4}{64}$
      \item $f(g(x))=\sqrt{\frac{x}{3}}$,
            $g(f(x))=\frac{\sqrt{x}}{3}$,
            $f(f(x))=\sqrt{\sqrt{x}}$ e
            $g(g(x))=\frac{x}{9}$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Dadas as funções \(f(x)=\frac{1}{x-4}\) e \(g(x)=x^2\),
  \begin{alphalist}
    \item Defina  $f(g(x))$  e $g(f(x))$ e seus domínios.
    \item Calcule $f(g(-3))$ e $g(f(7))$.
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $f(g(x))= \frac{1}{x^2-4}$ e
            $D(f(g))=\{ x \in \R| x \neq -2, x \neq 2 \}$,
            $g(f(x))=\frac{1}{(x-4)^2}$ e
            $D(g(f))=\{ x \in \R| x \neq 4\}$
      \item $f(g(-3))= \frac{1}{5}$ e
            $g(f(7))= \frac{1}{9}$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Na cidade de Salicilina, o número de casos de determinada doença viral
  varia ao longo do ano, sendo dado aproximadamente por
  \begin{multline*}
    d(t) = 1,9245t^4 - 53,485t^3 \\
         + 450,18t^2 - 1081,5t + 837,37,
  \end{multline*}
  em que $t$ é o mês do ano. Por sua vez, o custo em reais do tratamento
  dessa doença é dado pela função $c(x)=600x+12000$, em que $x$ é o número
  de pessoas infectadas.
  \begin{alphalist}
    \item Determine $c(d(t))$.
    \item Indique o que essa função composta representa.
    \item Calcule o custo aproximado de tratamento dos salicilinenses
          infectados no mês de abril. (Use uma calculadora).
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item $c(d(t)) = 1154,7t^4 - 32091t^3 + 270108t^2 - 648900t + 514422$
      \item O valor interno $d(t)$ calcula o número de casos para um determinado
            mês ($t$). O valor externo $c(d(t))$ calcula o custo total baseado
            nesse número de casos.
      \item $d(4) \approx 783,33$ pessoas. Como o número de pessoas é inteiro
            devemos calcular o custo para 784 pessoas.
            Assim $c(784) = \reais{482.400}$
    \end{alphalist}
  \end{answer}
\end{exercise}
