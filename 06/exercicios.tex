\begin{exercise}
  Considere as funções
  \( f(x) = \log_2 \sqrt[3]{x} \) e
  \( g(x) = \frac{-x^2+9}{x+2} \)
  \begin{alphalist}
    \item Encontre $f \circ g$
    \item Calcule  $f \circ g (-1)$
    \item Encontre o domínio da função $f \circ g$
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item \(f\circ g(x) = \log_2\sqrt[3]{\frac{-x^2+9}{x+2}}\)
      \item \(f\circ g(-1)
             = \log_2\sqrt[3]{\frac{8}{(-1)+2}}
             = \log_2\sqrt[3]{\frac{8}{1}}
             = \log_2 2
             = 1\)
      \item O domínio de $f\circ g(x)$ é $x\in (-\infty, -3) \cup (-2, 3)$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Considere a função $f(x)=\log_2(4-x^2)$ e responda o que se pede
  \begin{alphalist}
    \item Qual o domínio e a imagem de $f$?
    \item A equação $f(x)=1$ possui solução? Justifique sua resposta
    \item $f$ é injetora? Justifique sua resposta
    \item Quanto a simetria, $f$ é par, ímpar ou nem par, nem ímpar.
          Justifique sua resposta
  \end{alphalist}

  \begin{answer}
    \begin{alphalist}
      \item Domínio: $(-2, 2)$. Imagem: $(-\infty, \log_2 4)$
      \item Sim, a equação $f(x)=1$ possui solução.
            As soluções são $x=-\sqrt{2}$ ou $x=\sqrt{2}$
      \item Não, $f$ não é injetora, pois para todo $x$
            em $(-2,2)$ temos $f(x)=f(-x)$
      \item A função $f$ é par.
            $\log_2(4-x^2)=\log_2(4-(-x)^2)$, portanto $f(x)=f(-x)$
    \end{alphalist}
  \end{answer}
\end{exercise}

\begin{exercise}
  Um vírus de um computador se espalha segundo a função
  \[
    c(t) = \frac{6500}{1+a.2^{bt}}
  \]
  Em que $c(t)$ é o número de computadores infectados no instante $t$ (em horas),
  contando a partir do momento em que a infecção foi detectada. Sabendo que
  $c(0)=100$, $c(3)=500$ e que $\log_2 3=1,5850$, determine as constantes $a$ e
  $b$ e o número de computadores infectados para $t=6$

  \begin{answer}
    $a = 64$; $b = -0,805$; $c(6) \approx 2018,17$, como o número e
    computadores infectados é inteiro, o número de computadores
    infectados após 6 horas é 2019
  \end{answer}
\end{exercise}
